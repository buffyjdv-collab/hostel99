   215→function formatDate(dateStr?: string | null): string {
   216→  if (!dateStr) return '—'
   217→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   218→}
   219→
   220→function formatRelativeTime(dateStr: string): string {
   221→  const date = new Date(dateStr)
   222→  const now = new Date()
   223→  const diffMs = now.getTime() - date.getTime()
   224→  const diffMins = Math.floor(diffMs / 60000)
   225→  const diffHours = Math.floor(diffMs / 3600000)
   226→  const diffDays = Math.floor(diffMs / 86400000)
   227→
   228→  if (diffMins < 1) return 'Just now'
   229→  if (diffMins < 60) return `${diffMins}m ago`
   230→  if (diffHours < 24) return `${diffHours}h ago`
   231→  if (diffDays < 7) return `${diffDays}d ago`
   232→  return date.toLocaleDateString('en-IN', { month: 'short', day: 'numeric' })
   233→}

   329→function ComplaintDetailDialog({
   330→  complaint,
   331→  open,
   332→  onClose,
   333→  staff,
   334→  onUpdateStatus,
   335→  onAssignStaff,
   336→  onAddResolution,
   337→  loading,
   338→}: {
   339→  complaint: ComplaintData | null
   340→  open: boolean
   341→  onClose: () => void
   342→  staff: StaffInfo[]
   343→  onUpdateStatus: (id: string, status: ComplaintStatus, resolution?: string, rating?: number) => void
   344→  onAssignStaff: (id: string, staffId: string) => void
   345→  onAddResolution: (id: string, resolution: string, rating: number) => void
   346→  loading: boolean
   347→}) {
   348→  // Derive initial values from complaint
   349→  const initialAssignTo = complaint?.assignedToId || ''
   350→  const initialRating = complaint?.rating || 0
   351→
   352→  const [assignTo, setAssignTo] = useState(initialAssignTo)
   353→  const [newStatus, setNewStatus] = useState<ComplaintStatus | ''>('')
   354→  const [resolutionNotes, setResolutionNotes] = useState('')
   355→  const [resolutionRating, setResolutionRating] = useState(initialRating)
   356→
   357→  if (!complaint) return null
   358→
   359→  const statusCfg = STATUS_CONFIG[complaint.status]
   360→  const priorityCfg = PRIORITY_CONFIG[complaint.priority]
   361→  const categoryCfg = CATEGORY_CONFIG[complaint.category]
   362→
   363→  // Build timeline from status changes
   364→  const timeline: { status: ComplaintStatus; date: string; label: string }[] = [
   365→    { status: 'open', date: complaint.createdAt, label: `Complaint created by ${complaint.createdBy?.name || 'System'}` },
   366→  ]
   367→  if (complaint.assignedTo) {
   368→    timeline.push({ status: 'assigned', date: complaint.updatedAt, label: `Assigned to ${complaint.assignedTo.name}` })
   369→  }
   370→  if (['in_progress', 'resolved', 'closed'].includes(complaint.status)) {
   371→    timeline.push({ status: 'in_progress', date: complaint.updatedAt, label: 'Work started' })
   372→  }
   373→  if (['resolved', 'closed'].includes(complaint.status)) {
   374→    timeline.push({ status: 'resolved', date: complaint.updatedAt, label: 'Complaint resolved' })
   375→  }
   376→  if (complaint.status === 'closed') {
   377→    timeline.push({ status: 'closed', date: complaint.updatedAt, label: 'Complaint closed' })
   378→  }
   379→
   380→  return (
   381→    <Dialog open={open} onOpenChange={onClose}>
   382→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   383→        <DialogHeader>
   384→          <DialogTitle className="flex items-center gap-2">
   385→            <MessageSquareWarning className="w-5 h-5 text-emerald-600" />
   386→            {complaint.title}
   387→          </DialogTitle>
   388→          <DialogDescription>
   389→            Complaint #{complaint.id.slice(-8).toUpperCase()}
   390→          </DialogDescription>
   391→        </DialogHeader>
   392→
   393→        <div className="space-y-5">
   394→          {/* Status Badges Row */}
   395→          <div className="flex flex-wrap items-center gap-2">
   396→            <Badge variant="secondary" className={`${statusCfg.bgClass} ${statusCfg.textClass}`}>
   397→              <div className={`h-1.5 w-1.5 rounded-full ${statusCfg.dotClass} mr-1`} />
   398→              {statusCfg.label}
   399→            </Badge>
   400→            <Badge variant="secondary" className={`${priorityCfg.bgClass} ${priorityCfg.textClass}`}>
   401→              <div className={`h-1.5 w-1.5 rounded-full ${priorityCfg.dotClass} mr-1`} />
   402→              {priorityCfg.label}
   403→            </Badge>
   404→            <Badge variant="secondary" className={`${categoryCfg.bgClass} ${categoryCfg.textClass}`}>
   405→              <categoryCfg.icon className="w-3 h-3 mr-1" />
   406→              {categoryCfg.label}
   407→            </Badge>
   408→          </div>
   409→
   410→          {/* Description */}
   411→          <div className="bg-muted/50 rounded-lg p-4">
   412→            <p className="text-sm font-medium text-muted-foreground mb-1">Description</p>
   413→            <p className="text-sm whitespace-pre-wrap">{complaint.description}</p>
   414→          </div>
   415→
   416→          {/* Info Grid */}
   417→          <div className="grid grid-cols-2 gap-3 text-sm">
   418→            <div className="flex justify-between p-2 bg-muted/30 rounded">
   419→              <span className="text-muted-foreground">Tenant</span>
   420→              <span className="font-medium">{complaint.tenant.name}</span>
   421→            </div>
   422→            <div className="flex justify-between p-2 bg-muted/30 rounded">
   423→              <span className="text-muted-foreground">Room</span>
   424→              <span className="font-medium">{complaint.tenant.room?.number || '—'}</span>
   425→            </div>
   426→            <div className="flex justify-between p-2 bg-muted/30 rounded">
   427→              <span className="text-muted-foreground">Assigned To</span>

   429→            </div>

   458→                const TimelineIcon = cfg.icon

   467→                    <div className="pt-0.5">
   468→                      <p className="text-sm font-medium">{item.label}</p>

   470→                    </div>
   471→                  </div>
   472→                )
   473→              })}
   474→            </div>
   475→          </div>
   476→
   477→          {/* Actions */}
   478→          <div className="border-t pt-4 space-y-4">
   479→            {/* Assign to Staff */}
   480→            {complaint.status !== 'resolved' && complaint.status !== 'closed' && (
   481→              <div className="flex items-end gap-3">
   482→                <div className="flex-1 space-y-1.5">
   483→                  <Label className="text-xs">Assign to Staff</Label>
   484→                  <Select value={assignTo} onValueChange={setAssignTo}>
   485→                    <SelectTrigger><SelectValue placeholder="Select staff member" /></SelectTrigger>
   486→                    <SelectContent>
   487→                      {staff.map(s => (
   488→                        <SelectItem key={s.id} value={s.id}>{s.name}</SelectItem>
   489→                      ))}
   490→                    </SelectContent>
   491→                  </Select>
   492→                </div>
   493→                <Button
   494→                  size="sm"
   495→                  onClick={() => assignTo && onAssignStaff(complaint.id, assignTo)}
   496→                  disabled={!assignTo || loading}
   497→                  className="bg-emerald-600 hover:bg-emerald-700"
   498→                >
   499→                  {loading ? <Loader2 className="w-4 h-4 animate-spin" /> : <UserCheck className="w-4 h-4" />}
   500→                </Button>
   501→              </div>
   502→            )}
   503→
   504→            {/* Update Status */}
   505→            {complaint.status !== 'resolved' && complaint.status !== 'closed' && (
   506→              <div className="flex items-end gap-3">
   507→                <div className="flex-1 space-y-1.5">
   508→                  <Label className="text-xs">Update Status</Label>
   509→                  <Select value={newStatus} onValueChange={v => setNewStatus(v as ComplaintStatus)}>
   510→                    <SelectTrigger><SelectValue placeholder="Select new status" /></SelectTrigger>
   511→                    <SelectContent>
   512→                      {Object.entries(STATUS_CONFIG)
   513→                        .filter(([key]) => key !== complaint.status)
   514→                        .map(([key, cfg]) => (
   515→                          <SelectItem key={key} value={key}>{cfg.label}</SelectItem>
   516→                        ))}
   517→                    </SelectContent>
   518→                  </Select>
   519→                </div>
   520→                <Button
   521→                  size="sm"
   522→                  onClick={() => newStatus && onUpdateStatus(complaint.id, newStatus as ComplaintStatus)}
   523→                  disabled={!newStatus || loading}
   524→                  className="bg-emerald-600 hover:bg-emerald-700"
   525→                >
   526→                  {loading ? <Loader2 className="w-4 h-4 animate-spin" /> : 'Update'}
