   224→function formatCurrency(value: number | null | undefined): string {
   225→  if (value == null) return '—'
   226→  return '₹' + value.toLocaleString('en-IN', { minimumFractionDigits: 0, maximumFractionDigits: 2 })
   227→}
   228→
   229→function formatDate(dateStr: string | null | undefined): string {
   230→  if (!dateStr) return '—'
   231→  try {
   232→    return new Date(dateStr).toLocaleDateString('en-IN', {
   233→      day: '2-digit',
   234→      month: 'short',
   235→      year: 'numeric',
   236→    })
   237→  } catch {
   238→    return '—'
   239→  }
   240→}
   241→
   242→function isWarrantyExpiringSoon(dateStr: string | null | undefined): boolean {
   243→  if (!dateStr) return false
   244→  const expiry = new Date(dateStr)
   245→  const now = new Date()
   246→  const diffDays = (expiry.getTime() - now.getTime()) / (1000 * 60 * 60 * 24)
   247→  return diffDays > 0 && diffDays <= 90
   248→}
   249→
   250→function isWarrantyExpired(dateStr: string | null | undefined): boolean {
   251→  if (!dateStr) return false
   252→  return new Date(dateStr) < new Date()
   253→}
