   108→// ── Helpers ──────────────────────────────────────────────────────────────────
   109→
   110→function formatDate(date: string | Date | null) {
   111→  if (!date) return 'N/A'
   112→  return new Date(date).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   113→}
   114→
   115→function formatDateTime(date: string | Date) {
   116→  return new Date(date).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' })
   117→}
   118→
   119→function isExpired(expiryDate: string | null) {
   120→  if (!expiryDate) return false
   121→  return new Date(expiryDate) < new Date()
   122→}
