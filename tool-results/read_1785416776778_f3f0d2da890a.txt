   158→function formatCurrency(amount: number) {
   159→  return new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', maximumFractionDigits: 0 }).format(amount)
   160→}

   162→function formatDate(date: string | Date) {
   163→  return new Date(date).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
