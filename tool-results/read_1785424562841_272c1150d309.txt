   236→function formatCurrency(amount: number): string {
   237→  return new Intl.NumberFormat('en-IN', {
   238→    style: 'currency',
   239→    currency: 'INR',
   240→    minimumFractionDigits: 0,
   241→    maximumFractionDigits: 0,
   242→  }).format(amount)
   243→}
   244→
   245→function formatDate(dateStr?: string | null): string {
   246→  if (!dateStr) return '—'
   247→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   248→}
   249→
   250→function parseNum(val: string): number {
   251→  const n = parseFloat(val)
   252→  return isNaN(n) ? 0 : n
   253→}

   515→// ── Record Payment Dialog ────────────────────────────────────────────────────
   516→
   517→function RecordPaymentDialog({
   518→  open,
   519→  onClose,
   520→  tenants,
   521→  properties,
   522→  onSubmit,
   523→  loading,
   524→}: {
   525→  open: boolean
   526→  onClose: () => void
   527→  tenants: { id: string; name: string }[]
   528→  properties: PropertyInfo[]
   529→  onSubmit: (data: RecordPaymentFormData) => void
   530→  loading: boolean
   531→}) {
   532→  const [form, setForm] = useState<RecordPaymentFormData>({ ...EMPTY_FORM })
   533→  const resetForm = () => setForm({ ...EMPTY_FORM })
   534→
   535→  const computedTotal = useMemo(() => {
   536→    const rent = parseNum(form.rentAmount)
   537→    const elec = parseNum(form.electricity)
   538→    const wat = parseNum(form.water)
   539→    const wifi = parseNum(form.wifi)
   540→    const food = parseNum(form.food)
   541→    const laun = parseNum(form.laundry)
   542→    const park = parseNum(form.parking)
   543→    const other = parseNum(form.otherCharges)
   544→    const fine = parseNum(form.lateFine)
   545→    const disc = parseNum(form.discount)
   546→    const adv = parseNum(form.advanceAdjust)
   547→    return rent + elec + wat + wifi + food + laun + park + other + fine - disc - adv
   548→  }, [form])
   549→
   550→  const handleField = (field: keyof RecordPaymentFormData, value: string) => {
   551→    setForm(prev => ({ ...prev, [field]: value }))
   552→  }
   553→
   554→  const handleSubmit = () => {
   555→    onSubmit({ ...form, amount: String(computedTotal) })
   556→  }
   557→
   558→  const isValid = form.tenantId && form.propertyId && form.month && form.year && computedTotal > 0
   559→
   560→  return (
   561→    <Dialog open={open} onOpenChange={onClose}>
   562→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   563→        <DialogHeader>
   564→          <DialogTitle className="flex items-center gap-2">
   565→            <CreditCard className="w-5 h-5 text-emerald-600" />
   566→            Record Payment
   567→          </DialogTitle>
   568→          <DialogDescription>
   569→            Record a new rent payment with charges breakdown
   570→          </DialogDescription>
   571→        </DialogHeader>
   572→
   573→        <div className="space-y-6">
   574→          {/* Tenant & Property */}
   575→          <div className="grid grid-cols-2 gap-4">
   576→            <div className="space-y-2">
   577→              <Label>Tenant *</Label>
   578→              <Select value={form.tenantId} onValueChange={v => handleField('tenantId', v)}>
   579→                <SelectTrigger><SelectValue placeholder="Select tenant" /></SelectTrigger>
   580→                <SelectContent>
   581→                  {tenants.map(t => (
   582→                    <SelectItem key={t.id} value={t.id}>{t.name}</SelectItem>
   583→                  ))}
   584→                </SelectContent>
   585→              </Select>
   586→            </div>
   587→            <div className="space-y-2">
   588→              <Label>Property *</Label>
   589→              <Select value={form.propertyId} onValueChange={v => handleField('propertyId', v)}>
   590→                <SelectTrigger><SelectValue placeholder="Select property" /></SelectTrigger>
