   183→function formatCurrency(amount: number): string {
   184→  return new Intl.NumberFormat('en-IN', {
   185→    style: 'currency',
   186→    currency: 'INR',
   187→    minimumFractionDigits: 0,
   188→    maximumFractionDigits: 0,
   189→  }).format(amount)
   190→}
   191→
   192→function formatDate(dateStr?: string | null): string {
   193→  if (!dateStr) return '—'
   194→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   195→}
   196→
   197→function getAttendancePercentage(attendance: AttendanceData[]): number {
   198→  if (!attendance || attendance.length === 0) return 0
   199→  const presentDays = attendance.filter(a => a.status === 'present').length
   200→  const halfDays = attendance.filter(a => a.status === 'half_day').length * 0.5
   201→  return Math.round(((presentDays + halfDays) / attendance.length) * 100)
   202→}
