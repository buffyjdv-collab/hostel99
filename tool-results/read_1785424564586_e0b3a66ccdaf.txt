   155→function parseAmenities(amenitiesStr?: string | null): string[] {
   156→  if (!amenitiesStr) return []
   157→  try {
   158→    const parsed = JSON.parse(amenitiesStr)
   159→    return Array.isArray(parsed) ? parsed : []
   160→  } catch {
   161→    return amenitiesStr.split(',').map((a) => a.trim()).filter(Boolean)
   162→  }
   163→}

   281→    return { totalProperties, totalRooms, totalBeds, avgOccupancy }

   287→      result = result.filter((p) => p.type === typeFilter)

   290→      const q = searchQuery.toLowerCase()
   291→      result = result.filter(

   293→          p.name.toLowerCase().includes(q) ||
   294→          p.address.toLowerCase().includes(q) ||
   295→          p.city.toLowerCase().includes(q) ||

   326→          ...formData,
   327→          ownerId: currentUser?.id || 'default-owner',
   328→          amenities: formData.amenities.length > 0 ? JSON.stringify(formData.amenities) : null,
   329→        }),

   334→        setProperties((prev) => [newProperty, ...prev])
   335→        setDialogOpen(false)
   336→        resetForm()

   338→    } catch (error) {
   339→      console.error('Failed to create property:', error)

   341→      setSubmitting(false)
   342→    }

   589→            <Filter className="h-4 w-4 text-muted-foreground" />
   590→            {FILTER_OPTIONS.map((opt) => (

   598→                    ? 'bg-emerald-600 hover:bg-emerald-700 text-white'

   601→              >
   602→                {opt.label}
   603→              </Button>
   604→            ))}
   605→          </div>
   606→          <div className="flex items-center gap-2">
   607→            <div className="relative">

   609→              <Input
   610→                placeholder="Search properties..."
   611→                value={searchQuery}
   612→                onChange={(e) => setSearchQuery(e.target.value)}
   613→                className="pl-9 w-64"
   614→              />
   615→            </div>
   616→            <div className="flex items-center rounded-md border">
   617→              <Button
   618→                variant="ghost"
   619→                size="sm"
   620→                className={viewMode === 'grid' ? 'bg-accent' : ''}
   621→                onClick={() => setViewMode('grid')}

   623→                <LayoutGrid className="h-4 w-4" />
   624→              </Button>

   626→                variant="ghost"

   656→              </p>
   657→              {!searchQuery && typeFilter === 'all' && (
   658→                <Button
   659→                  className="mt-4 bg-emerald-600 hover:bg-emerald-700 text-white"
   660→                  onClick={() => setDialogOpen(true)}

   662→                  <Plus className="mr-2 h-4 w-4" />

   665→              )}

   667→          </Card>

   669→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">

   671→              <PropertyCard key={property.id} property={property} />

   673→          </div>

   675→          <div className="space-y-3">

   677→              <PropertyListItem key={property.id} property={property} />

   681→      </div>
   682→    </ScrollArea>
