   158→function PurposeBadge({ purpose }: { purpose: string | null }) {
   159→  const config: Record<string, { label: string; className: string }> = {
   160→    breakfast: { label: 'Breakfast', className: 'bg-amber-100 text-amber-800 dark:bg-amber-900/30 dark:text-amber-300' },
   161→    lunch: { label: 'Lunch', className: 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-300' },
   162→    dinner: { label: 'Dinner', className: 'bg-purple-100 text-purple-800 dark:bg-purple-900/30 dark:text-purple-300' },
   163→    snacks: { label: 'Snacks', className: 'bg-orange-100 text-orange-800 dark:bg-orange-900/30 dark:text-orange-300' },
   164→    special: { label: 'Special', className: 'bg-rose-100 text-rose-800 dark:bg-rose-900/30 dark:text-rose-300' },
   165→  }
   166→  const c = config[purpose || ''] || { label: purpose || 'General', className: 'bg-slate-100 text-slate-800 dark:bg-slate-800 dark:text-slate-300' }
   167→  return <Badge variant="secondary" className={c.className}>{c.label}</Badge>
   168→}

   172→function CategoryBadge({ category }: { category: string }) {
   173→  const config: Record<string, { label: string; icon: React.ReactNode; className: string }> = {
   174→    veg: { label: 'Veg', icon: <Leaf className="h-3 w-3" />, className: 'bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-300' },

   176→    vegan: { label: 'Vegan', icon: <Apple className="h-3 w-3" />, className: 'bg-emerald-100 text-emerald-800 dark:bg-emerald-900/30 dark:text-emerald-300' },
   177→  }
   178→  const c = config[category] || { label: category, icon: null, className: 'bg-slate-100 text-slate-800 dark:bg-slate-800 dark:text-slate-300' }

   197→function MealTypeIcon({ mealType, className }: { mealType: string; className?: string }) {
   198→  switch (mealType) {
   199→    case 'breakfast': return <Soup className={className || 'h-4 w-4'} />
   200→    case 'lunch': return <Utensils className={className || 'h-4 w-4'} />
   201→    case 'dinner': return <UtensilsCrossed className={className || 'h-4 w-4'} />
   202→    case 'snacks': return <Apple className={className || 'h-4 w-4'} />
   203→    default: return <Utensils className={className || 'h-4 w-4'} />
   204→  }
   205→}
