   210→function parseAmenities(amenitiesStr?: string | null): string[] {
   211→  if (!amenitiesStr) return []
   212→  try {
   213→    const parsed = JSON.parse(amenitiesStr)
   214→    return Array.isArray(parsed) ? parsed : []
   215→  } catch {
   216→    return amenitiesStr.split(',').map((a) => a.trim()).filter(Boolean)
   217→  }
   218→}

   344→export function RoomsPage() {

   347→  // ── State ────────────────────────────────────────────────────────────────

   355→  const [submitting, setSubmitting] = useState(false)
   356→
   357→  // Filters
   358→  const [filterProperty, setFilterProperty] = useState('all')
   359→  const [filterFloor, setFilterFloor] = useState('all')
   360→  const [filterRoomType, setFilterRoomType] = useState('all')
   361→  const [filterSharingType, setFilterSharingType] = useState('all')
   362→  const [filterStatus, setFilterStatus] = useState('all')
   363→
   364→  // Form
   365→  const [formData, setFormData] = useState<RoomFormData>({
   366→    propertyId: '',
   367→    buildingId: '',
   368→    floorId: '',
   369→    name: '',
   370→    number: '',
   371→    sharingType: 'single',
   372→    roomType: 'non_ac',

   374→    rent: 0,
   375→    deposit: 0,
   376→    amenities: [],
   377→  })
   378→
   379→  // ── Derived Data ─────────────────────────────────────────────────────────
   380→  const selectedProperty = useMemo(
   381→    () => properties.find((p) => p.id === formData.propertyId),
   382→    [properties, formData.propertyId]
   383→  )
   384→
   385→  const availableBuildings = useMemo(() => selectedProperty?.buildings ?? [], [selectedProperty])
   386→
   387→  const selectedBuilding = useMemo(
   388→    () => availableBuildings.find((b) => b.id === formData.buildingId),
   389→    [availableBuildings, formData.buildingId]
   390→  )
   391→
   392→  const availableFloors = useMemo(() => selectedBuilding?.floors ?? [], [selectedBuilding])
   393→
   394→  // Floor options for filter
   395→  const floorOptions = useMemo(() => {
   396→    const floors = new Set<string>()
   397→    rooms.forEach((r) => {
   398→      if (r.floor?.name) floors.add(r.floor.name)
   399→    })
   400→    return Array.from(floors).sort()
   401→  }, [rooms])
   402→
   403→  // ── Data Fetching ────────────────────────────────────────────────────────
   404→  useEffect(() => {
   405→    Promise.all([fetchRooms(), fetchProperties()])
   406→  }, [])
   407→
   408→  async function fetchRooms() {
   409→    try {
   410→      setLoading(true)
   411→      const res = await fetch('/api/rooms')
   412→      if (res.ok) {
   413→        const data = await res.json()
   414→        setRooms(data)
   415→      }
   416→    } catch (error) {
   417→      console.error('Failed to fetch rooms:', error)
   418→    } finally {
   419→      setLoading(false)
   420→    }
   421→  }
   422→
   423→  async function fetchProperties() {
   424→    try {
   425→      const res = await fetch('/api/properties')
   426→      if (res.ok) {
   427→        setProperties(await res.json())
   428→      }
   429→    } catch (error) {
   430→      console.error('Failed to fetch properties:', error)
   431→    }
   432→  }
   433→
   434→  // ── Computed Values ──────────────────────────────────────────────────────
   435→  const stats = useMemo(() => {
   436→    const totalRooms = rooms.length
   437→    const totalBeds = rooms.reduce((sum, r) => sum + r.totalBeds, 0)
   438→    const occupied = rooms.reduce((sum, r) => sum + r.occupiedBedsCount, 0)
   439→    const vacant = totalBeds - occupied
   440→    const occupancyPct = totalBeds > 0 ? Math.round((occupied / totalBeds) * 100) : 0
   441→    return { totalRooms, totalBeds, occupied, vacant, occupancyPct }
   442→  }, [rooms])
   443→
   444→  const filteredRooms = useMemo(() => {
   445→    let result = rooms
   446→    if (filterProperty !== 'all') {
   447→      result = result.filter((r) => r.propertyId === filterProperty)
   448→    }
   449→    if (filterFloor !== 'all') {
   450→      result = result.filter((r) => r.floor?.name === filterFloor)
   451→    }
   452→    if (filterRoomType !== 'all') {
   453→      result = result.filter((r) => r.roomType === filterRoomType)
   454→    }
   455→    if (filterSharingType !== 'all') {
   456→      result = result.filter((r) => r.sharingType === filterSharingType)
   457→    }
   458→    if (filterStatus !== 'all') {
   459→      result = result.filter((r) => r.status === filterStatus)
   460→    }

   462→      const q = searchQuery.toLowerCase()
   463→      result = result.filter(
   464→        (r) =>
   465→          r.name.toLowerCase().includes(q) ||
   466→          r.number.toLowerCase().includes(q) ||
   467→          r.property?.name.toLowerCase().includes(q) ||
   468→          r.building?.name.toLowerCase().includes(q)
   469→      )
   470→    }
   471→    return result
   472→  }, [rooms, filterProperty, filterFloor, filterRoomType, filterSharingType, filterStatus, searchQuery])

   474→  // ── Form Handlers ────────────────────────────────────────────────────────
   475→  function handleFormChange(field: keyof RoomFormData, value: string | number | string[]) {
   476→    setFormData((prev) => {
   477→      const next = { ...prev, [field]: value }
   478→      // Reset building when property changes
   479→      if (field === 'propertyId') {
   480→        next.buildingId = ''
   481→        next.floorId = ''
   482→      }
   483→      // Reset floor when building changes
   484→      if (field === 'buildingId') {
   485→        next.floorId = ''
   486→      }
   487→      return next
   488→    })
   489→  }

   491→  function handleAmenityToggle(amenity: string) {
   492→    setFormData((prev) => ({
   493→      ...prev,
   494→      amenities: prev.amenities.includes(amenity)
   495→        ? prev.amenities.filter((a) => a !== amenity)
   496→        : [...prev.amenities, amenity],
   497→    }))
   498→  }

   500→  async function handleSubmit(e: React.FormEvent) {
   501→    e.preventDefault()
   502→    if (!formData.name || !formData.number || !formData.floorId || !formData.buildingId || !formData.propertyId) return
   503→
   504→    try {
   505→      setSubmitting(true)
   506→      const res = await fetch('/api/rooms', {
   507→        method: 'POST',
   508→        headers: { 'Content-Type': 'application/json' },
   509→        body: JSON.stringify({
   510→          ...formData,
   511→          amenities: formData.amenities.length > 0 ? JSON.stringify(formData.amenities) : null,
   512→          status: 'available',
   513→        }),
   514→      })
   515→
   516→      if (res.ok) {
   517→        const newRoom = await res.json()
   518→        setRooms((prev) => [newRoom, ...prev])
   519→        setDialogOpen(false)
   520→        resetForm()
   521→      }
   522→    } catch (error) {
   523→      console.error('Failed to create room:', error)
   524→    } finally {
   525→      setSubmitting(false)
   526→    }
   527→  }

   556→  // ── Render ───────────────────────────────────────────────────────────────
