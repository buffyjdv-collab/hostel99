   173→function formatCurrency(amount: number): string {
   174→  return new Intl.NumberFormat('en-IN', {
   175→    style: 'currency',
   176→    currency: 'INR',
   177→    minimumFractionDigits: 0,
   178→    maximumFractionDigits: 0,
   179→  }).format(amount)
   180→}
   181→
   182→function formatDate(dateStr?: string | null): string {
   183→  if (!dateStr) return '—'
   184→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   185→}
   186→
   187→function isOverdue(dateStr?: string | null): boolean {
   188→  if (!dateStr) return false
   189→  return new Date(dateStr) < new Date()
   190→}

   280→  lead,
   281→  open,
   282→  onClose,
   283→  onStageChange,
   284→}: {
   285→  lead: LeadData | null
   286→  open: boolean
   287→  onClose: () => void
   288→  onStageChange: (leadId: string, newStage: LeadStage) => void
   289→}) {
   290→  if (!lead) return null
   291→

   293→  const canMoveForward = currentIdx < 6
   294→  const canMoveBack = currentIdx > 0
   295→
   296→  const timeline = PIPELINE_STAGES.map((s, idx) => ({
   297→    ...s,
   298→    completed: idx < currentIdx,
   299→    current: idx === currentIdx,
   300→  }))
   301→
   302→  const sourceCfg = SOURCE_CONFIG[lead.source]
   303→  const SourceIcon = sourceCfg.icon
   304→
   305→  return (
   306→    <Dialog open={open} onOpenChange={onClose}>

   314→            </Badge>
   315→          </DialogTitle>
   316→          <DialogDescription>
   317→            Lead details and pipeline stage management
   318→          </DialogDescription>
   319→        </DialogHeader>
   320→
   321→        {/* Pipeline Timeline */}
   322→        <div className="space-y-2">
   323→          <h4 className="text-sm font-semibold text-slate-700 dark:text-slate-300">Pipeline Progress</h4>

   340→                        <div
   341→                          className={`h-0.5 w-1 ${
