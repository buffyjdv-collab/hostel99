   236→function formatCurrency(amount: number): string {
   237→  return new Intl.NumberFormat('en-IN', {
   238→    style: 'currency',
   239→    currency: 'INR',
   240→    minimumFractionDigits: 0,
   241→    maximumFractionDigits: 0,
   242→  }).format(amount)
   243→}
   244→
   245→function formatDate(dateStr?: string | null): string {
   246→  if (!dateStr) return '—'
   247→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   248→}
   249→
   250→function parseNum(val: string): number {
   251→  const n = parseFloat(val)
   252→  return isNaN(n) ? 0 : n
   253→}

   515→// ── Record Payment Dialog ────────────────────────────────────────────────────
   516→
   517→function RecordPaymentDialog({
   518→  open,
   519→  onClose,
   520→  tenants,
   521→  properties,
   522→  onSubmit,
   523→  loading,
   524→}: {
   525→  open: boolean
   526→  onClose: () => void
   527→  tenants: { id: string; name: string }[]
   528→  properties: PropertyInfo[]
   529→  onSubmit: (data: RecordPaymentFormData) => void
   530→  loading: boolean
   531→}) {
   532→  const [form, setForm] = useState<RecordPaymentFormData>({ ...EMPTY_FORM })
   533→
   534→  // Reset form when dialog opens
   535→  useEffect(() => {
   536→    if (open) setForm({ ...EMPTY_FORM })
   537→  }, [open])
   538→
   539→  const computedTotal = useMemo(() => {
   540→    const rent = parseNum(form.rentAmount)
   541→    const elec = parseNum(form.electricity)
   542→    const wat = parseNum(form.water)
   543→    const wifi = parseNum(form.wifi)
   544→    const food = parseNum(form.food)
   545→    const laun = parseNum(form.laundry)
   546→    const park = parseNum(form.parking)
   547→    const other = parseNum(form.otherCharges)
   548→    const fine = parseNum(form.lateFine)
   549→    const disc = parseNum(form.discount)
   550→    const adv = parseNum(form.advanceAdjust)
   551→    return rent + elec + wat + wifi + food + laun + park + other + fine - disc - adv
   552→  }, [form])
   553→
   554→  const handleField = (field: keyof RecordPaymentFormData, value: string) => {
   555→    setForm(prev => ({ ...prev, [field]: value }))
   556→  }
   557→
   558→  const handleSubmit = () => {
   559→    onSubmit({ ...form, amount: String(computedTotal) })
   560→  }
   561→
   562→  const isValid = form.tenantId && form.propertyId && form.month && form.year && computedTotal > 0
   563→
   564→  return (
   565→    <Dialog open={open} onOpenChange={onClose}>
   566→      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   567→        <DialogHeader>
   568→          <DialogTitle className="flex items-center gap-2">
   569→            <CreditCard className="w-5 h-5 text-emerald-600" />
   570→            Record Payment
   571→          </DialogTitle>
   572→          <DialogDescription>
   573→            Record a new rent payment with charges breakdown
   574→          </DialogDescription>
   575→        </DialogHeader>
   576→
   577→        <div className="space-y-6">
   578→          {/* Tenant & Property */}
   579→          <div className="grid grid-cols-2 gap-4">
   580→            <div className="space-y-2">
   581→              <Label>Tenant *</Label>
   582→              <Select value={form.tenantId} onValueChange={v => handleField('tenantId', v)}>
   583→                <SelectTrigger><SelectValue placeholder="Select tenant" /></SelectTrigger>
   584→                <SelectContent>
   585→                  {tenants.map(t => (
   586→                    <SelectItem key={t.id} value={t.id}>{t.name}</SelectItem>
   587→                  ))}
   588→                </SelectContent>
   589→              </Select>
   590→            </div>
   591→            <div className="space-y-2">
   592→              <Label>Property *</Label>
   593→              <Select value={form.propertyId} onValueChange={v => handleField('propertyId', v)}>
