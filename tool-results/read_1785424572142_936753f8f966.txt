   169→function StockReport({ selectedPropertyId }: { selectedPropertyId: string }) {
   170→  const [invData, setInvData] = useState<any>(null)
   171→  const [invLoading, setInvLoading] = useState(true)
   172→
   173→  useEffect(() => {
   174→    const fetchInventory = async () => {
   175→      try {
   176→        const pid = selectedPropertyId || ''
   177→        const res = await fetch(`/api/inventory?propertyId=${pid}`)
   178→        const data = await res.json()
   179→        setInvData(data)
   180→      } catch (e) { console.error(e) }
   181→      finally { setInvLoading(false) }
   182→    }
   183→    fetchInventory()
   184→  }, [selectedPropertyId])
   185→
   186→  if (invLoading) return <div className="flex items-center justify-center p-12"><Loader2 className="h-8 w-8 animate-spin text-emerald-500" /></div>
   187→  if (!invData) return <div className="text-center text-muted-foreground p-12">No data available</div>
   188→
   189→  const items = invData.items || []
   190→  const categories = invData.categories || []
   191→  const stats = invData.stats || {}
   192→
   193→  const categoryData = categories.map((c: any) => {
   194→    const catItems = items.filter((i: any) => i.categoryId === c.id)
   195→    return { name: c.name, count: catItems.length, value: catItems.reduce((s: number, i: any) => s + i.currentStock * i.unitPrice, 0) }
   196→  }).filter((c: any) => c.count > 0)
   197→
   198→  return (
   199→    <div className="space-y-6">
   200→      <div className="grid gap-4 md:grid-cols-4">
   201→        <Card><CardContent className="p-4"><p className="text-sm text-muted-foreground">Total Items</p><p className="text-2xl font-bold">{stats.totalItems || 0}</p></CardContent></Card>
   202→        <Card><CardContent className="p-4"><p className="text-sm text-muted-foreground">Low Stock</p><p className="text-2xl font-bold text-red-500">{stats.lowStockCount || 0}</p></CardContent></Card>
   203→        <Card><CardContent className="p-4"><p className="text-sm text-muted-foreground">Total Value</p><p className="text-2xl font-bold">{formatCurrency(stats.totalValue || 0)}</p></CardContent></Card>
   204→        <Card><CardContent className="p-4"><p className="text-sm text-muted-foreground">Expiring Soon</p><p className="text-2xl font-bold text-amber-500">{stats.expiringSoon || 0}</p></CardContent></Card>
   205→      </div>
   206→      {categoryData.length > 0 && (
   207→        <Card>
   208→          <CardHeader><CardTitle>Stock Value by Category</CardTitle></CardHeader>
   209→          <CardContent>
   210→            <ChartContainer config={{ value: { label: 'Value', color: '#10b981' } }} className="h-[300px]">
   211→              <BarChart data={categoryData}>
   212→                <CartesianGrid strokeDasharray="3 3" />
   213→                <XAxis dataKey="name" tick={{ fontSize: 12 }} />
   214→                <YAxis tickFormatter={(v) => `₹${(v / 1000).toFixed(0)}k`} />
   215→                <ChartTooltip content={<ChartTooltipContent />} />
   216→                <Bar dataKey="value" fill="#10b981" radius={[4, 4, 0, 0]} />
   217→              </BarChart>
   218→            </ChartContainer>
   219→          </CardContent>
   220→        </Card>
   221→      )}
   222→      <Card>
   223→        <CardHeader><CardTitle>Current Stock Levels</CardTitle></CardHeader>
   224→        <CardContent>
   225→          <div className="overflow-x-auto">
   226→            <Table>
   227→              <TableHeader><TableRow><TableHead>Item</TableHead><TableHead>Category</TableHead><TableHead>Unit</TableHead><TableHead>Current Stock</TableHead><TableHead>Min Stock</TableHead><TableHead>Unit Price</TableHead><TableHead>Value</TableHead><TableHead>Status</TableHead></TableRow></TableHeader>
   228→              <TableBody>
   229→                {items.map((item: any) => (
   230→                  <TableRow key={item.id}>
   231→                    <TableCell className="font-medium">{item.name}</TableCell>
   232→                    <TableCell><Badge variant="outline">{item.category?.name || '-'}</Badge></TableCell>
   233→                    <TableCell>{item.unit}</TableCell>

   235→                    <TableCell>{item.minStock}</TableCell>
   236→                    <TableCell>{formatCurrency(item.unitPrice)}</TableCell>
   237→                    <TableCell>{formatCurrency(item.currentStock * item.unitPrice)}</TableCell>

   280→        <Card><CardContent className="p-4"><p className="text-sm text-muted-foreground">Total Wastage</p><p className="text-2xl font-bold text-red-500">{totalWastage.toFixed(1)}</p></CardContent></Card>
   281→        <Card><CardContent className="p-4"><p className="text-sm text-muted-foreground">Total Cost</p><p className="text-2xl font-bold">{formatCurrency(totalCost)}</p></CardContent></Card>

   284→        <CardHeader><CardTitle>Consumption Log</CardTitle></CardHeader>

   291→                  <TableRow key={log.id}>
   292→                    <TableCell>{format(new Date(log.date), 'dd MMM')}</TableCell>
   293→                    <TableCell className="font-medium">{log.item?.name || '-'}</TableCell>

   303→              </TableBody>
   304→            </Table>
   305→          </div>
   306→        </CardContent>
   307→      </Card>
   308→    </div>
   309→  )

   391→function KitchenCostReport({ selectedPropertyId }: { selectedPropertyId: string }) {

   393→  const [kitchenData, setKitchenData] = useState<any>(null)

   396→  useEffect(() => {
   397→    const fetchData = async () => {
   398→      try {
   399→        const pid = selectedPropertyId || ''
   400→        const [invRes, kitchenRes] = await Promise.all([
   401→          fetch(`/api/inventory?propertyId=${pid}`),
   402→          fetch(`/api/kitchen?propertyId=${pid}&type=issues`),
   403→        ])
   404→        setInvData(await invRes.json())
   405→        setKitchenData(await kitchenRes.json())
   406→      } catch (e) { console.error(e) }
   407→      finally { setDataLoading(false) }

   409→    fetchData()

   412→  if (dataLoading) return <div className="flex items-center justify-center p-12"><Loader2 className="h-8 w-8 animate-spin text-emerald-500" /></div>
   413→
   414→  const totalStockValue = invData?.stats?.totalValue || 0
   415→  const issues = kitchenData?.issues || []
   416→  const totalIssuedCost = issues.reduce((s: number, i: any) => s + (i.quantity * (i.item?.unitPrice || 0)), 0)
   417→  const tenants = 80 // approximate
   418→  const costPerStudent = tenants > 0 ? totalIssuedCost / tenants : 0
   419→
   420→  return (
   421→    <div className="space-y-6">

   587→function ExpiryReport({ selectedPropertyId }: { selectedPropertyId: string }) {
   588→  const [invData, setInvData] = useState<any>(null)
   589→  const [invLoading, setInvLoading] = useState(true)
   590→
   591→  useEffect(() => {
   592→    const fetchInventory = async () => {
   593→      try {
   594→        const pid = selectedPropertyId || ''
   595→        const res = await fetch(`/api/inventory?propertyId=${pid}`)
   596→        const data = await res.json()
   597→        setInvData(data)
   598→      } catch (e) { console.error(e) }
   599→      finally { setInvLoading(false) }
   600→    }
   601→    fetchInventory()
   602→  }, [selectedPropertyId])
   603→
   604→  if (invLoading) return <div className="flex items-center justify-center p-12"><Loader2 className="h-8 w-8 animate-spin text-emerald-500" /></div>
   605→
   606→  const now = new Date()
   607→  const items = (invData?.items || []).filter((i: any) => i.expiryDate).map((i: any) => ({
   608→    ...i,
   609→    daysUntilExpiry: Math.ceil((new Date(i.expiryDate).getTime() - now.getTime()) / (1000 * 60 * 60 * 24)),
   610→  })).sort((a: any, b: any) => a.daysUntilExpiry - b.daysUntilExpiry)
   611→
   612→  return (
   613→    <div className="space-y-6">
   614→      <Card>
   615→        <CardHeader><CardTitle>Items with Expiry Dates</CardTitle></CardHeader>
   616→        <CardContent>
   617→          <div className="overflow-x-auto">
   618→            <Table>
   619→              <TableHeader><TableRow><TableHead>Item</TableHead><TableHead>Category</TableHead><TableHead>Batch</TableHead><TableHead>Current Stock</TableHead><TableHead>Expiry Date</TableHead><TableHead>Days Left</TableHead><TableHead>Status</TableHead></TableRow></TableHeader>
   620→              <TableBody>
   621→                {items.map((item: any) => (
   622→                  <TableRow key={item.id} className={item.daysUntilExpiry <= 0 ? 'bg-red-50/50' : item.daysUntilExpiry <= 7 ? 'bg-amber-50/50' : ''}>
   623→                    <TableCell className="font-medium">{item.name}</TableCell>
   624→                    <TableCell><Badge variant="outline">{item.category?.name || '-'}</Badge></TableCell>
   625→                    <TableCell>{item.batchNumber || '-'}</TableCell>
   626→                    <TableCell>{item.currentStock} {item.unit}</TableCell>
   627→                    <TableCell>{format(new Date(item.expiryDate), 'dd MMM yyyy')}</TableCell>
   628→                    <TableCell className={item.daysUntilExpiry <= 0 ? 'text-red-600 font-bold' : item.daysUntilExpiry <= 7 ? 'text-amber-600 font-semibold' : ''}>{item.daysUntilExpiry} days</TableCell>
   629→                    <TableCell><Badge variant={item.daysUntilExpiry <= 0 ? 'destructive' : item.daysUntilExpiry <= 7 ? 'secondary' : 'outline'}>{item.daysUntilExpiry <= 0 ? 'Expired' : item.daysUntilExpiry <= 7 ? 'Expiring Soon' : 'OK'}</Badge></TableCell>
   630→                  </TableRow>
   631→                ))}
   632→                {items.length === 0 && <TableRow><TableCell colSpan={7} className="text-center text-muted-foreground py-8">No items with expiry dates</TableCell></TableRow>}
   633→              </TableBody>
   634→            </Table>
   635→          </div>
   636→        </CardContent>
   637→      </Card>
   638→    </div>
