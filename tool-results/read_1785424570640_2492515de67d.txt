   220→function formatCurrency(amount: number): string {
   221→  return new Intl.NumberFormat('en-IN', {
   222→    style: 'currency',
   223→    currency: 'INR',
   224→    minimumFractionDigits: 0,
   225→    maximumFractionDigits: 0,
   226→  }).format(amount)
   227→}
   228→
   229→function formatDate(dateStr?: string | null): string {
   230→  if (!dateStr) return '—'
   231→  try {
   232→    return format(new Date(dateStr), 'dd MMM yyyy')
   233→  } catch {
   234→    return '—'
   235→  }
   236→}
   237→
   238→function formatDateTime(dateStr?: string | null): string {
   239→  if (!dateStr) return '—'
   240→  try {
   241→    return format(new Date(dateStr), 'dd MMM yyyy, hh:mm a')
   242→  } catch {
   243→    return '—'
   244→  }
   245→}

   247→function getStockStatus(current: number, min: number): { label: string; color: string; bgClass: string; textClass: string } {
   248→  if (current <= 0) {
   249→    return { label: 'Out of Stock', color: 'bg-red-500', bgClass: 'bg-red-100 dark:bg-red-950/50', textClass: 'text-red-700 dark:text-red-300' }
   250→  }
   251→  if (current <= min) {
   252→    return { label: 'Low Stock', color: 'bg-red-500', bgClass: 'bg-red-100 dark:bg-red-950/50', textClass: 'text-red-700 dark:text-red-300' }
   253→  }
   254→  if (current <= min * 1.5) {
   255→    return { label: 'Near Min', color: 'bg-amber-500', bgClass: 'bg-amber-100 dark:bg-amber-950/50', textClass: 'text-amber-700 dark:text-amber-300' }
   256→  }
   257→  return { label: 'In Stock', color: 'bg-emerald-500', bgClass: 'bg-emerald-100 dark:bg-emerald-950/50', textClass: 'text-emerald-700 dark:text-emerald-300' }
   258→}
   259→
   260→function isExpiringSoon(expiryDate: string | null): boolean {
   261→  if (!expiryDate) return false
   262→  const sevenDaysFromNow = new Date(Date.now() + 7 * 24 * 60 * 60 * 1000)
   263→  return new Date(expiryDate) <= sevenDaysFromNow
   264→}
   265→
   266→function isExpired(expiryDate: string | null): boolean {
   267→  if (!expiryDate) return false
   268→  return new Date(expiryDate) < new Date()
   269→}

  1082→  // ── Computed Values ────────────────────────────────────────────────────────

  1084→  const sortedItems = useMemo(() => {
