   106→const PAYMENT_TERMS = [
   107→  { value: 'COD', label: 'Cash on Delivery' },

   109→  { value: 'Net 30', label: 'Net 30 Days' },

   116→function getStatusBadge(status: string) {
   117→  switch (status) {
   118→    case 'active':
   119→      return <Badge className="bg-emerald-100 text-emerald-800 dark:bg-emerald-900/30 dark:text-emerald-400 border-0"><CheckCircle2 className="h-3 w-3 mr-1" />Active</Badge>
   120→    case 'inactive':
   121→      return <Badge className="bg-slate-100 text-slate-600 dark:bg-slate-800 dark:text-slate-400 border-0"><XCircle className="h-3 w-3 mr-1" />Inactive</Badge>
   122→    case 'blacklisted':
   123→      return <Badge className="bg-red-100 text-red-800 dark:bg-red-900/30 dark:text-red-400 border-0"><Ban className="h-3 w-3 mr-1" />Blacklisted</Badge>
   124→    default:
   125→      return <Badge variant="outline">{status}</Badge>
   126→  }
   127→}
   128→
   129→function renderStars(rating: number) {
   130→  return (
   131→    <div className="flex items-center gap-0.5">
   132→      {[1, 2, 3, 4, 5].map((star) => (
   133→        <Star
   134→          key={star}
   135→          className={`h-3.5 w-3.5 ${star <= rating ? 'text-amber-400 fill-amber-400' : 'text-slate-300 dark:text-slate-600'}`}
   136→        />
   137→      ))}
   138→      <span className="text-xs text-slate-500 ml-1">{rating.toFixed(1)}</span>
   139→    </div>
   140→  )
   141→}

   146→  const { selectedPropertyId, currentUser } = useAppStore()

   148→

   152→  const [search, setSearch] = useState('')
   153→  const [statusFilter, setStatusFilter] = useState<string>('all')
   154→

   156→  const [addDialogOpen, setAddDialogOpen] = useState(false)
   157→  const [editDialogOpen, setEditDialogOpen] = useState(false)

   160→

   164→    contactPerson: '',

   167→    address: '',
   168→    city: '',
   169→    state: '',
   170→    pincode: '',
   171→    gstNumber: '',
   172→    panNumber: '',
   173→    bankName: '',
   174→    bankAccount: '',

   178→  })

   213→        (v.contactPerson || '').toLowerCase().includes(q) ||
   214→        (v.city || '').toLowerCase().includes(q) ||
   215→        (v.email || '').toLowerCase().includes(q) ||

   224→
   225→  // ── Form Handlers ──────────────────────────────────────────────
   226→

   229→      name: '', contactPerson: '', phone: '', email: '', address: '',
   230→      city: '', state: '', pincode: '', gstNumber: '', panNumber: '',
   231→      bankName: '', bankAccount: '', ifscCode: '', paymentTerms: 'Net 30', rating: 0,
   232→    })

   234→

   245→        body: JSON.stringify({ ...form, propertyId: selectedPropertyId }),

   250→        resetForm()
   251→        fetchVendors()

   253→        toast({ title: 'Error', description: 'Failed to add vendor', variant: 'destructive' })
   254→      }
   255→    } catch {
   256→      toast({ title: 'Error', description: 'Failed to add vendor', variant: 'destructive' })
   257→    } finally {
   258→      setSubmitting(false)
   259→    }
   260→  }
   261→
   262→  const handleEdit = async () => {
   263→    if (!selectedVendor) return

   266→      const res = await fetch('/api/vendors', {
   267→        method: 'PATCH',

   272→        toast({ title: 'Success', description: 'Vendor updated successfully' })
   273→        setEditDialogOpen(false)
   274→        fetchVendors()

   276→        toast({ title: 'Error', description: 'Failed to update vendor', variant: 'destructive' })
   277→      }
   278→    } catch {
   279→      toast({ title: 'Error', description: 'Failed to update vendor', variant: 'destructive' })
   280→    } finally {
   281→      setSubmitting(false)
   282→    }
   283→  }

   288→        method: 'PATCH',
   289→        headers: { 'Content-Type': 'application/json' },

   295→      }
   296→    } catch {
   297→      toast({ title: 'Error', description: 'Failed to update status', variant: 'destructive' })
   298→    }
   299→  }

   305→      contactPerson: vendor.contactPerson || '',
   306→      phone: vendor.phone,
   307→      email: vendor.email || '',
   308→      address: vendor.address || '',
   309→      city: vendor.city || '',
   310→      state: vendor.state || '',
   311→      pincode: vendor.pincode || '',
   312→      gstNumber: vendor.gstNumber || '',
   313→      panNumber: vendor.panNumber || '',
   314→      bankName: vendor.bankName || '',
   315→      bankAccount: vendor.bankAccount || '',
   316→      ifscCode: vendor.ifscCode || '',
   317→      paymentTerms: vendor.paymentTerms || 'Net 30',
   318→      rating: vendor.rating,
   319→    })
   320→    setEditDialogOpen(true)
   321→  }
   322→
   323→  const openDetail = (vendor: VendorData) => {

   327→
   328→  // ── Render ─────────────────────────────────────────────────────
   329→

   331→    return (
   332→      <div className="space-y-6">
   333→        <Skeleton className="h-8 w-48" />
   334→        <div className="grid gap-4 md:grid-cols-3">
   335→          <Skeleton className="h-24 rounded-lg" />
   336→          <Skeleton className="h-24 rounded-lg" />
   337→          <Skeleton className="h-24 rounded-lg" />
   338→        </div>
   339→        <Skeleton className="h-[400px] rounded-lg" />
   340→      </div>
   341→    )
   342→  }
   343→
   344→  return (
   345→    <div className="space-y-6">
   346→      {/* Header */}
   347→      <div className="flex items-center justify-between">
   348→        <div>
   349→          <h1 className="text-2xl font-bold text-slate-900 dark:text-slate-100">Vendor Management</h1>
   350→          <p className="text-slate-500 dark:text-slate-400 mt-1">Manage suppliers, track purchase history, and vendor performance</p>
   351→        </div>
   352→        <Button onClick={() => { resetForm(); setAddDialogOpen(true) }} className="bg-emerald-600 hover:bg-emerald-700">
   353→          <Plus className="h-4 w-4 mr-2" /> Add Vendor
   354→        </Button>
   355→      </div>
   356→
   357→      {/* Stats */}
   358→      <div className="grid gap-4 md:grid-cols-3">
   359→        <Card>
   360→          <CardContent className="p-4">
   361→            <div className="flex items-center gap-3">
   362→              <div className="p-2 rounded-lg bg-blue-100 dark:bg-blue-900/30">
   363→                <Truck className="h-5 w-5 text-blue-600 dark:text-blue-400" />
   364→              </div>
   365→              <div>
   366→                <div className="text-sm text-slate-500 dark:text-slate-400">Total Vendors</div>
   367→                <div className="text-2xl font-bold text-slate-900 dark:text-slate-100">{stats.totalVendors}</div>
   368→              </div>
   369→            </div>
   370→          </CardContent>
   371→        </Card>
   372→        <Card>
   373→          <CardContent className="p-4">
   374→            <div className="flex items-center gap-3">
   375→              <div className="p-2 rounded-lg bg-emerald-100 dark:bg-emerald-900/30">
   376→                <CheckCircle2 className="h-5 w-5 text-emerald-600 dark:text-emerald-400" />
   377→              </div>
   378→              <div>
   379→                <div className="text-sm text-slate-500 dark:text-slate-400">Active Vendors</div>
   380→                <div className="text-2xl font-bold text-emerald-600 dark:text-emerald-400">{stats.activeVendors}</div>
   381→              </div>
   382→            </div>
   383→          </CardContent>
   384→        </Card>
   385→        <Card>
   386→          <CardContent className="p-4">
   387→            <div className="flex items-center gap-3">
   388→              <div className="p-2 rounded-lg bg-amber-100 dark:bg-amber-900/30">
   389→                <Star className="h-5 w-5 text-amber-500 dark:text-amber-400" />
   390→              </div>
   391→              <div>
   392→                <div className="text-sm text-slate-500 dark:text-slate-400">Average Rating</div>
   393→                <div className="text-2xl font-bold text-amber-600 dark:text-amber-400">{stats.avgRating} / 5</div>
   394→              </div>
   395→            </div>
   396→          </CardContent>
   397→        </Card>
   398→      </div>
   399→
   400→      {/* Filters */}
   401→      <div className="flex flex-col sm:flex-row gap-3">
   402→        <div className="relative flex-1">
   403→          <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-slate-400" />
   404→          <Input
   405→            placeholder="Search vendors by name, contact, city, GST..."
   406→            value={search}
   407→            onChange={(e) => setSearch(e.target.value)}
   408→            className="pl-10"
   409→          />
   410→        </div>
   411→        <Select value={statusFilter} onValueChange={setStatusFilter}>
   412→          <SelectTrigger className="w-[160px]">
   413→            <SelectValue placeholder="Status" />
   414→          </SelectTrigger>
   415→          <SelectContent>
   416→            <SelectItem value="all">All Status</SelectItem>
   417→            <SelectItem value="active">Active</SelectItem>
   418→            <SelectItem value="inactive">Inactive</SelectItem>
   419→            <SelectItem value="blacklisted">Blacklisted</SelectItem>
   420→          </SelectContent>
   421→        </Select>
   422→      </div>
   423→
   424→      {/* Table */}
   425→      <Card>
   426→        <CardContent className="p-0">
   427→          <div className="overflow-x-auto">
   428→            <Table>
   429→              <TableHeader>
   430→                <TableRow>
   431→                  <TableHead>Vendor</TableHead>
   432→                  <TableHead className="hidden md:table-cell">Contact Person</TableHead>
   433→                  <TableHead>Phone</TableHead>
   434→                  <TableHead className="hidden lg:table-cell">Email</TableHead>
   435→                  <TableHead className="hidden xl:table-cell">GST Number</TableHead>

   437→                  <TableHead>Rating</TableHead>

   453→                  </TableRow>
   454→                ) : (
   455→                  filteredVendors.map((vendor) => (
   456→                    <TableRow key={vendor.id} className="hover:bg-slate-50 dark:hover:bg-slate-800/50 cursor-pointer" onClick={() => openDetail(vendor)}>
   457→                      <TableCell>
   458→                        <div className="flex items-center gap-3">
   459→                          <div className="w-8 h-8 rounded-full bg-blue-100 dark:bg-blue-900/30 flex items-center justify-center text-blue-600 dark:text-blue-400 font-semibold text-sm">
   460→                            {vendor.name.charAt(0)}
   461→                          </div>
   462→                          <div>
   463→                            <div className="font-medium text-slate-900 dark:text-slate-100">{vendor.name}</div>
   464→                            {vendor.paymentTerms && (
   465→                              <div className="text-xs text-slate-400">{vendor.paymentTerms}</div>
   466→                            )}
   467→                          </div>
   468→                        </div>
   469→                      </TableCell>
   470→                      <TableCell className="hidden md:table-cell text-slate-600 dark:text-slate-400">{vendor.contactPerson || '-'}</TableCell>
   471→                      <TableCell>
   472→                        <div className="flex items-center gap-1.5">
   473→                          <Phone className="h-3 w-3 text-slate-400" />
   474→                          <span>{vendor.phone}</span>
   475→                        </div>
   476→                      </TableCell>
   477→                      <TableCell className="hidden lg:table-cell">
   478→                        {vendor.email ? (
   479→                          <div className="flex items-center gap-1.5">
   480→                            <Mail className="h-3 w-3 text-slate-400" />
   481→                            <span className="text-slate-600 dark:text-slate-400">{vendor.email}</span>
   482→                          </div>
   483→                        ) : '-'}
   484→                      </TableCell>
   485→                      <TableCell className="hidden xl:table-cell font-mono text-xs text-slate-500">{vendor.gstNumber || '-'}</TableCell>
   486→                      <TableCell className="hidden md:table-cell">
   487→                        {vendor.city ? (

   489→                            <MapPin className="h-3 w-3 text-slate-400" />
   490→                            <span>{vendor.city}</span>

   493→                      </TableCell>

   499→                      <TableCell>
   500→                        <DropdownMenu>

   502→                            <Button variant="ghost" size="sm" className="h-8 w-8 p-0">
   503→                              <MoreHorizontal className="h-4 w-4" />
   504→                            </Button>
   505→                          </DropdownMenuTrigger>
   506→                          <DropdownMenuContent align="end">
   507→                            <DropdownMenuItem onClick={(e) => { e.stopPropagation(); openDetail(vendor) }}>

   510→                            <DropdownMenuItem onClick={(e) => { e.stopPropagation(); openEdit(vendor) }}>
   511→                              <Edit className="h-4 w-4 mr-2" /> Edit
   512→                            </DropdownMenuItem>
   513→                            <DropdownMenuItem onClick={(e) => { e.stopPropagation() }}>
   514→                              <ShoppingCart className="h-4 w-4 mr-2" /> Create PO

   518→                                <XCircle className="h-4 w-4 mr-2" /> Deactivate

   521→                            {vendor.status === 'inactive' && (
   522→                              <DropdownMenuItem onClick={(e) => { e.stopPropagation(); handleStatusChange(vendor, 'active') }} className="text-emerald-600">
   523→                                <CheckCircle2 className="h-4 w-4 mr-2" /> Activate
   524→                              </DropdownMenuItem>
   525→                            )}
   526→                            {vendor.status !== 'blacklisted' && (
   527→                              <DropdownMenuItem onClick={(e) => { e.stopPropagation(); handleStatusChange(vendor, 'blacklisted') }} className="text-red-600">
   528→                                <Ban className="h-4 w-4 mr-2" /> Blacklist
   529→                              </DropdownMenuItem>
   530→                            )}
   531→                          </DropdownMenuContent>
   532→                        </DropdownMenu>
   533→                      </TableCell>
   534→                    </TableRow>
   535→                  ))
   536→                )}
   537→              </TableBody>
   538→            </Table>
