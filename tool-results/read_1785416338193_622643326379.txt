   199→// ─── Helpers ───────────────────────────────────────────────────
   200→function formatCurrency(amount: number) {
   201→  return `₹${amount.toLocaleString('en-IN', { minimumFractionDigits: 0, maximumFractionDigits: 2 })}`
   202→}
   203→
   204→function formatDate(dateStr: string) {
   205→  try {
   206→    return format(new Date(dateStr), 'dd MMM yyyy')
   207→  } catch {
   208→    return dateStr
   209→  }
   210→}
   211→
   212→function getMealColor(mealType: string) {
   213→  return MEAL_TYPES.find(m => m.value === mealType) || MEAL_TYPES[0]
   214→}
   215→
   216→function getWasteCategoryLabel(cat: string) {
   217→  return WASTE_CATEGORIES.find(c => c.value === cat)?.label || cat
   218→}
   219→
   220→function getDisposalLabel(method: string) {
   221→  return DISPOSAL_METHODS.find(d => d.value === method)?.label || method
   222→}

   226→  const { selectedPropertyId, currentUser } = useAppStore()
   227→  const { toast } = useToast()

   232→  const [messData, setMessData] = useState<MessData>({})
   233→
   234→  // Attendance state
   235→  const [attendanceDate, setAttendanceDate] = useState<Date>(new Date())
   236→  const [selectedMeal, setSelectedMeal] = useState<string>('breakfast')
   237→  const [tenants, setTenants] = useState<Tenant[]>([])

   246→
   247→  // Consumption state
   248→  const [consumptionDays, setConsumptionDays] = useState(7)
   249→  const [consumptionSearch, setConsumptionSearch] = useState('')
   250→  const [showConsumptionDialog, setShowConsumptionDialog] = useState(false)
   251→  const [savingConsumption, setSavingConsumption] = useState(false)
   252→  const [inventoryItems, setInventoryItems] = useState<InventoryItem[]>([])
   253→
   254→  // Waste state
   255→  const [wasteDays, setWasteDays] = useState(7)
   256→  const [showWasteDialog, setShowWasteDialog] = useState(false)
   257→  const [savingWaste, setSavingWaste] = useState(false)
   258→
   259→  // Consumption form
   260→  const [consumptionForm, setConsumptionForm] = useState({
   261→    itemId: '',
   262→    mealType: 'breakfast',
   263→    issuedQty: 0,
   264→    consumedQty: 0,
   265→    returnedQty: 0,
   266→    wastageQty: 0,
   267→    date: format(new Date(), 'yyyy-MM-dd'),
   268→    notes: '',
   269→  })
   270→
   271→  // Waste form
   272→  const [wasteForm, setWasteForm] = useState({
   273→    category: 'food_waste',
   274→    description: '',
   275→    itemId: '',
   276→    quantity: 0,
   277→    unit: 'kg',
   278→    estimatedCost: 0,
   279→    disposalMethod: 'trash',
   280→    date: format(new Date(), 'yyyy-MM-dd'),
   281→    notes: '',
   282→  })
   283→
   284→  // ─── Data fetching ─────────────────────────────────────────
   285→  const fetchMessData = useCallback(async (type: string = 'all') => {
   286→    setLoading(true)
   287→    try {
   288→      const params = new URLSearchParams()
   289→      if (selectedPropertyId) params.set('propertyId', selectedPropertyId)
   290→      params.set('type', type)
   291→      if (type === 'all' || type === 'attendance') {
   292→        params.set('date', format(attendanceDate, 'yyyy-MM-dd'))
   293→      }
   294→      if (type === 'all' || type === 'consumption') {
   295→        params.set('days', String(consumptionDays))
   296→      }
   297→      if (type === 'all' || type === 'waste') {
   298→        params.set('days', String(wasteDays))
   299→      }
   300→
   301→      const res = await fetch(`/api/mess?${params.toString()}`)
   302→      if (!res.ok) throw new Error('Failed to fetch mess data')
   303→      const data = await res.json()
   304→      setMessData(prev => ({ ...prev, ...data }))
   305→    } catch (error) {
   306→      console.error('Error fetching mess data:', error)
   307→      toast({ title: 'Error', description: 'Failed to fetch mess data', variant: 'destructive' })
   308→    } finally {
   309→      setLoading(false)
   310→    }
   311→  }, [selectedPropertyId, attendanceDate, consumptionDays, wasteDays, toast])
   312→
   313→  const fetchTenants = useCallback(async () => {
   314→    try {
   315→      const params = new URLSearchParams()
   316→      if (selectedPropertyId) params.set('propertyId', selectedPropertyId)
   317→      const res = await fetch(`/api/tenants?${params.toString()}`)
   318→      if (!res.ok) throw new Error('Failed to fetch tenants')
   319→      const data = await res.json()
   320→      const tenantList = Array.isArray(data) ? data : []
   321→      setTenants(tenantList)
   322→      // Initialize attendance rows
   323→      setAttendanceRows(
   324→        tenantList.map((t: Tenant) => ({
   325→          tenantId: t.id,
   326→          name: t.user?.name || t.name,
   327→          present: true,
   328→          guestCount: 0,
   329→          notes: '',
   330→        }))
   331→      )
   332→    } catch (error) {
   333→      console.error('Error fetching tenants:', error)
   334→    }
   335→  }, [selectedPropertyId])
   336→
   337→  const fetchInventoryItems = useCallback(async () => {
   338→    try {
   339→      const params = new URLSearchParams()
   340→      if (selectedPropertyId) params.set('propertyId', selectedPropertyId)
   341→      const res = await fetch(`/api/inventory?${params.toString()}`)
   342→      if (!res.ok) throw new Error('Failed to fetch inventory')
   343→      const data = await res.json()
   344→      setInventoryItems(Array.isArray(data.items) ? data.items : [])
   345→    } catch (error) {
   346→      console.error('Error fetching inventory:', error)
   347→    }
   348→  }, [selectedPropertyId])
   349→
   350→  // ─── Effects ───────────────────────────────────────────────
   351→  useEffect(() => {
   352→    fetchMessData(activeTab)
   353→  }, [activeTab, attendanceDate, consumptionDays, wasteDays, fetchMessData])
   354→
   355→  useEffect(() => {
   356→    fetchTenants()
   357→  }, [selectedPropertyId, fetchTenants])
   358→
   359→  useEffect(() => {
   360→    fetchInventoryItems()
   361→  }, [selectedPropertyId, fetchInventoryItems])
   362→
   363→  // Load existing attendance records into the rows when data changes
   364→  useEffect(() => {
   365→    if (messData.attendance && tenants.length > 0) {
   366→      const mealData = messData.attendance[selectedMeal]
   367→      if (mealData?.records?.length) {
   368→        setAttendanceRows(prev =>
   369→          prev.map(row => {
   370→            const existing = mealData.records.find(
   371→              (r: AttendanceRecord) => r.tenantId === row.tenantId
   372→            )
   373→            if (existing) {
   374→              return {
   375→                ...row,
   376→                present: existing.present,
   377→                guestCount: existing.guestCount,
   378→                notes: existing.notes || '',
   379→              }
   380→            }
   381→            return row
   382→          })
   383→        )
   384→      }
   385→    }
   386→  }, [messData.attendance, selectedMeal, tenants.length])
   387→
   388→  // ─── Computed values ───────────────────────────────────────
   389→  const attendanceStats = useMemo(() => {
   390→    const mealData = messData.attendance?.[selectedMeal]
   391→    if (!mealData) {
   392→      return { totalPresent: 0, totalGuests: 0, attendanceRate: 0, totalTenants: tenants.length }
   393→    }
   394→    const totalPresent = mealData.present
   395→    const totalGuests = mealData.guests
   396→    const total = mealData.present + mealData.absent
   397→    const rate = total > 0 ? Math.round((totalPresent / total) * 100) : 0
   398→    return { totalPresent, totalGuests, attendanceRate: rate, totalTenants: total }
   399→  }, [messData.attendance, selectedMeal, tenants.length])
   400→
   401→  const filteredConsumption = useMemo(() => {
   402→    const logs = messData.consumption || []
   403→    if (!consumptionSearch) return logs
   404→    const q = consumptionSearch.toLowerCase()
   405→    return logs.filter(
   406→      (c: ConsumptionLog) =>
   407→        c.item?.name?.toLowerCase().includes(q) ||
   408→        c.mealType?.toLowerCase().includes(q)
   409→    )
   410→  }, [messData.consumption, consumptionSearch])
   411→
   412→  const consumptionTotals = useMemo(() => {
   413→    return filteredConsumption.reduce(
   414→      (acc: { issued: number; consumed: number; returned: number; wastage: number; cost: number }, c: ConsumptionLog) => ({
   415→        issued: acc.issued + c.issuedQty,
   416→        consumed: acc.consumed + c.consumedQty,
   417→        returned: acc.returned + c.returnedQty,
   418→        wastage: acc.wastage + c.wastageQty,
   419→        cost: acc.cost + c.totalCost,
   420→      }),
   421→      { issued: 0, consumed: 0, returned: 0, wastage: 0, cost: 0 }
   422→    )
   423→  }, [filteredConsumption])
   424→
   425→  const consumptionChartData = useMemo(() => {
   426→    if (!messData.consumption?.length) return []
   427→    const grouped: Record<string, { consumed: number; wastage: number; label: string }> = {}
   428→    for (const log of messData.consumption) {
   429→      try {
   430→        const dayKey = format(new Date(log.date), 'dd MMM')
   431→        if (!grouped[dayKey]) grouped[dayKey] = { consumed: 0, wastage: 0, label: dayKey }
   432→        grouped[dayKey].consumed += log.consumedQty
   433→        grouped[dayKey].wastage += log.wastageQty
   434→      } catch {
   435→        // skip invalid dates
   436→      }
   437→    }
   438→    return Object.values(grouped).sort((a, b) => a.label.localeCompare(b.label))
   439→  }, [messData.consumption])
   440→
   441→  const wasteStats = useMemo(() => {
   442→    return messData.wasteStats || { totalWaste: 0, foodWaste: 0, expired: 0, damaged: 0 }
   443→  }, [messData.wasteStats])
   444→
   445→  // ─── Handlers ──────────────────────────────────────────────

   450→      const res = await fetch('/api/mess', {
   451→        method: 'POST',

   454→          type: 'attendance',
   455→          propertyId: selectedPropertyId,

   461→            present: r.present,
   462→            guestCount: r.guestCount,
   463→            notes: r.notes,

   465→        }),
   466→      })

   478→      setSavingAttendance(false)
   479→    }
   480→  }

   486→    }

   491→
   492→      const res = await fetch('/api/mess', {
   493→        method: 'POST',
   494→        headers: { 'Content-Type': 'application/json' },
   495→        body: JSON.stringify({
   496→          type: 'consumption',
   497→          propertyId: selectedPropertyId,
   498→          date: consumptionForm.date,
   499→          itemId: consumptionForm.itemId,
   500→          mealType: consumptionForm.mealType,
   501→          issuedQty: consumptionForm.issuedQty,
   502→          consumedQty: consumptionForm.consumedQty,
   503→          returnedQty: consumptionForm.returnedQty,
   504→          wastageQty: consumptionForm.wastageQty,
   505→          unit: selectedItem?.unit || 'kg',
   506→          costPerUnit: selectedItem?.unitPrice || 0,
   507→          totalCost,
   508→          notes: consumptionForm.notes,
   509→          userId: currentUser?.id,
   510→        }),
   511→      })
   512→      if (!res.ok) throw new Error('Failed to add consumption log')
   513→      toast({ title: 'Success', description: 'Consumption log added successfully' })
   514→      setShowConsumptionDialog(false)
   515→      setConsumptionForm({
   516→        itemId: '',
   517→        mealType: 'breakfast',
   518→        issuedQty: 0,
   519→        consumedQty: 0,
   520→        returnedQty: 0,
   521→        wastageQty: 0,
   522→        date: format(new Date(), 'yyyy-MM-dd'),
   523→        notes: '',
   524→      })
   525→      fetchMessData('consumption')
   526→    } catch (error) {
   527→      console.error('Error adding consumption:', error)
   528→      toast({ title: 'Error', description: 'Failed to add consumption log', variant: 'destructive' })
   529→    } finally {
   530→      setSavingConsumption(false)
   531→    }
   532→  }
   533→
   534→  const handleAddWaste = async () => {
   535→    if (!wasteForm.description) {
   536→      toast({ title: 'Validation Error', description: 'Please enter a description', variant: 'destructive' })
   537→      return
   538→    }
   539→    setSavingWaste(true)
   540→    try {
   541→      const res = await fetch('/api/mess', {
   542→        method: 'POST',
   543→        headers: { 'Content-Type': 'application/json' },
   544→        body: JSON.stringify({
   545→          type: 'waste',
   546→          propertyId: selectedPropertyId,
   547→          date: wasteForm.date,
   548→          category: wasteForm.category,
   549→          itemId: wasteForm.itemId || null,
   550→          description: wasteForm.description,
   551→          quantity: wasteForm.quantity,
   552→          unit: wasteForm.unit,
   553→          estimatedCost: wasteForm.estimatedCost,
   554→          disposalMethod: wasteForm.disposalMethod,
   555→          notes: wasteForm.notes,
   556→          userId: currentUser?.id,
   557→        }),
   558→      })
   559→      if (!res.ok) throw new Error('Failed to add waste record')
   560→      toast({ title: 'Success', description: 'Waste record added successfully' })
   561→      setShowWasteDialog(false)
   562→      setWasteForm({
   563→        category: 'food_waste',
   564→        description: '',
   565→        itemId: '',
   566→        quantity: 0,
   567→        unit: 'kg',
   568→        estimatedCost: 0,
   569→        disposalMethod: 'trash',
   570→        date: format(new Date(), 'yyyy-MM-dd'),
   571→        notes: '',
   572→      })
   573→      fetchMessData('waste')
   574→    } catch (error) {
   575→      console.error('Error adding waste:', error)
