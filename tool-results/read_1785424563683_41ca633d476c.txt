   297→function formatCurrency(amount: number): string {
   298→  return new Intl.NumberFormat('en-IN', {
   299→    style: 'currency',
   300→    currency: 'INR',
   301→    minimumFractionDigits: 0,
   302→    maximumFractionDigits: 0,
   303→  }).format(amount)
   304→}
   305→
   306→function formatDate(dateStr: string | null | undefined): string {
   307→  if (!dateStr) return '—'
   308→  const d = new Date(dateStr)
   309→  return d.toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   310→}
