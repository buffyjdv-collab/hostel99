   218→function formatCurrency(amount: number): string {
   219→  return new Intl.NumberFormat('en-IN', {
   220→    style: 'currency',
   221→    currency: 'INR',
   222→    minimumFractionDigits: 0,
   223→    maximumFractionDigits: 0,
   224→  }).format(amount)
   225→}
   226→
   227→function formatDate(dateStr?: string | null): string {
   228→  if (!dateStr) return '—'
   229→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   230→}
