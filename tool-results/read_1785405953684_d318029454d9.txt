   197→function formatCurrency(amount: number): string {
   198→  return new Intl.NumberFormat('en-IN', {
   199→    style: 'currency',
   200→    currency: 'INR',
   201→    maximumFractionDigits: 0,
   202→  }).format(amount)
   203→}
   204→
   205→function formatDate(date: Date): string {
   206→  return date.toLocaleDateString('en-IN', {
   207→    weekday: 'long',
   208→    year: 'numeric',
   209→    month: 'long',
   210→    day: 'numeric',
   211→  })
   212→}
   213→
   214→function formatRelativeTime(dateStr: string): string {
   215→  const date = new Date(dateStr)
   216→  const now = new Date()
   217→  const diffMs = now.getTime() - date.getTime()
   218→  const diffMins = Math.floor(diffMs / 60000)
   219→  const diffHours = Math.floor(diffMs / 3600000)
   220→  const diffDays = Math.floor(diffMs / 86400000)
   221→
   222→  if (diffMins < 1) return 'Just now'
   223→  if (diffMins < 60) return `${diffMins}m ago`
   224→  if (diffHours < 24) return `${diffHours}h ago`
   225→  if (diffDays < 7) return `${diffDays}d ago`
   226→  return date.toLocaleDateString('en-IN', { month: 'short', day: 'numeric' })
   227→}

   229→function getComplaintStatusIcon(status: string) {
   230→  switch (status) {
   231→    case 'open':
   232→      return <CircleDot className="w-4 h-4 text-amber-500" />
   233→    case 'in_progress':
   234→      return <Clock className="w-4 h-4 text-blue-500" />
   235→    case 'resolved':
   236→      return <CheckCircle2 className="w-4 h-4 text-emerald-500" />
   237→    case 'closed':
   238→      return <XCircle className="w-4 h-4 text-slate-400" />
   239→    default:
   240→      return <CircleDot className="w-4 h-4 text-slate-400" />
   241→  }
   242→}
   243→
   244→function getComplaintStatusColor(status: string): string {
   245→  switch (status) {
   246→    case 'open': return 'bg-amber-50 text-amber-700 border-amber-200'
   247→    case 'in_progress': return 'bg-blue-50 text-blue-700 border-blue-200'
   248→    case 'resolved': return 'bg-emerald-50 text-emerald-700 border-emerald-200'
   249→    case 'closed': return 'bg-slate-50 text-slate-600 border-slate-200'
   250→    default: return 'bg-slate-50 text-slate-600 border-slate-200'
   251→  }
   252→}

   255→  switch (status) {

   258→    case 'resolved': return 'Resolved'
   259→    case 'closed': return 'Closed'

   266→interface KPICardProps {

   275→function KPICard({ title, value, icon: Icon, trend, iconBg = 'bg-emerald-50', iconColor = 'text-emerald-600' }: KPICardProps) {
   276→  return (
   277→    <Card className="hover:shadow-md transition-shadow">
   278→      <CardContent className="pt-0">
   279→        <div className="flex items-start justify-between">
   280→          <div className="space-y-1">
   281→            <p className="text-sm text-muted-foreground font-medium">{title}</p>
   282→            <p className="text-2xl font-bold tracking-tight">{value}</p>
   283→            {trend && (
   284→              <div className="flex items-center gap-1 text-xs">
   285→                {trend.value >= 0 ? (
   286→                  <TrendingUp className="w-3 h-3 text-emerald-500" />
   287→                ) : (
   288→                  <TrendingDown className="w-3 h-3 text-red-500" />
   289→                )}
   290→                <span className={trend.value >= 0 ? 'text-emerald-600' : 'text-red-600'}>
   291→                  {trend.value >= 0 ? '+' : ''}{trend.value}%
   292→                </span>
   293→                <span className="text-muted-foreground">{trend.label}</span>
   294→              </div>
   295→            )}
   296→          </div>
   297→          <div className={`flex items-center justify-center w-10 h-10 rounded-lg ${iconBg}`}>
   298→            <Icon className={`w-5 h-5 ${iconColor}`} />
   299→          </div>
   300→        </div>
   301→      </CardContent>
   302→    </Card>
   303→  )
   304→}
