   178→
   179→function formatCurrency(amount: number) {
   180→  return new Intl.NumberFormat('en-IN', { style: 'currency', currency: 'INR', maximumFractionDigits: 0 }).format(amount)
   181→}
   182→
   183→function formatDate(date: string | Date | null | undefined) {
   184→  if (!date) return 'Never'
   185→  return new Date(date).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   186→}
   187→
   188→function formatDateTime(date: string | Date | null | undefined) {
   189→  if (!date) return 'Never'
   190→  return new Date(date).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric', hour: '2-digit', minute: '2-digit' })
   191→}
