   200→function formatCurrency(amount: number): string {
   201→  return new Intl.NumberFormat('en-IN', {
   202→    style: 'currency',
   203→    currency: 'INR',
   204→    minimumFractionDigits: 0,
   205→    maximumFractionDigits: 0,
   206→  }).format(amount)
   207→}
   208→
   209→function formatDate(dateStr?: string | null): string {
   210→  if (!dateStr) return '—'
   211→  return new Date(dateStr).toLocaleDateString('en-IN', { day: '2-digit', month: 'short', year: 'numeric' })
   212→}
