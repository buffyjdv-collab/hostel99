   196→// ─── Helpers ───────────────────────────────────────────────────
   197→function formatCurrency(amount: number) {
   198→  return `₹${amount.toLocaleString('en-IN', { minimumFractionDigits: 0, maximumFractionDigits: 2 })}`
   199→}
   200→
   201→function formatDate(dateStr: string) {
   202→  try {
   203→    return format(new Date(dateStr), 'dd MMM yyyy')
   204→  } catch {
   205→    return dateStr
   206→  }
   207→}
   208→
   209→function getMealColor(mealType: string) {
   210→  return MEAL_TYPES.find(m => m.value === mealType) || MEAL_TYPES[0]
   211→}
   212→
   213→function getWasteCategoryLabel(cat: string) {
   214→  return WASTE_CATEGORIES.find(c => c.value === cat)?.label || cat
   215→}
   216→
   217→function getDisposalLabel(method: string) {
   218→  return DISPOSAL_METHODS.find(d => d.value === method)?.label || method
   219→}

   221→// ─── Component ─────────────────────────────────────────────────
   222→export function MessPage() {
   223→  const { selectedPropertyId, currentUser } = useAppStore()
   224→  const { toast } = useToast()
   225→
   226→  // ─── State ─────────────────────────────────────────────────
   227→  const [activeTab, setActiveTab] = useState('attendance')
   228→  const [loading, setLoading] = useState(true)
   229→  const [messData, setMessData] = useState<MessData>({})
   230→
   231→  // Attendance state
   232→  const [attendanceDate, setAttendanceDate] = useState<Date>(new Date())
   233→  const [selectedMeal, setSelectedMeal] = useState<string>('breakfast')
   234→  const [tenants, setTenants] = useState<Tenant[]>([])
   235→  const [attendanceRows, setAttendanceRows] = useState<{
   236→    tenantId: string
   237→    name: string
   238→    present: boolean
   239→    guestCount: number
   240→    notes: string
   241→  }[]>([])
   242→  const [savingAttendance, setSavingAttendance] = useState(false)
   243→
   244→  // Consumption state
   245→  const [consumptionDays, setConsumptionDays] = useState(7)
   246→  const [consumptionSearch, setConsumptionSearch] = useState('')
   247→  const [showConsumptionDialog, setShowConsumptionDialog] = useState(false)
   248→  const [savingConsumption, setSavingConsumption] = useState(false)
   249→  const [inventoryItems, setInventoryItems] = useState<InventoryItem[]>([])
   250→
   251→  // Waste state
   252→  const [wasteDays, setWasteDays] = useState(7)
   253→  const [showWasteDialog, setShowWasteDialog] = useState(false)
   254→  const [savingWaste, setSavingWaste] = useState(false)
   255→
   256→  // Consumption form
   257→  const [consumptionForm, setConsumptionForm] = useState({
   258→    itemId: '',
   259→    mealType: 'breakfast',
   260→    issuedQty: 0,
   261→    consumedQty: 0,
   262→    returnedQty: 0,
   263→    wastageQty: 0,
   264→    date: format(new Date(), 'yyyy-MM-dd'),
   265→    notes: '',
   266→  })
   267→
   268→  // Waste form
   269→  const [wasteForm, setWasteForm] = useState({
   270→    category: 'food_waste',
   271→    description: '',
   272→    itemId: '',
   273→    quantity: 0,
   274→    unit: 'kg',
   275→    estimatedCost: 0,
   276→    disposalMethod: 'trash',
   277→    date: format(new Date(), 'yyyy-MM-dd'),
   278→    notes: '',
   279→  })
   280→
   281→  // ─── Data fetching ─────────────────────────────────────────
   282→  const fetchMessData = useCallback(async (type: string = 'all') => {
   283→    setLoading(true)
   284→    try {
   285→      const params = new URLSearchParams()
   286→      if (selectedPropertyId) params.set('propertyId', selectedPropertyId)
   287→      params.set('type', type)
   288→      if (type === 'all' || type === 'attendance') {
   289→        params.set('date', format(attendanceDate, 'yyyy-MM-dd'))
   290→      }
   291→      if (type === 'all' || type === 'consumption') {
   292→        params.set('days', String(consumptionDays))
   293→      }
   294→      if (type === 'all' || type === 'waste') {
   295→        params.set('days', String(wasteDays))
   296→      }
   297→
   298→      const res = await fetch(`/api/mess?${params.toString()}`)
   299→      if (!res.ok) throw new Error('Failed to fetch mess data')
   300→      const data = await res.json()
   301→      setMessData(prev => ({ ...prev, ...data }))
   302→    } catch (error) {
   303→      console.error('Error fetching mess data:', error)
   304→      toast({ title: 'Error', description: 'Failed to fetch mess data', variant: 'destructive' })
   305→    } finally {
   306→      setLoading(false)
   307→    }
   308→  }, [selectedPropertyId, attendanceDate, consumptionDays, wasteDays, toast])
   309→
   310→  const fetchTenants = useCallback(async () => {
   311→    try {
   312→      const params = new URLSearchParams()
   313→      if (selectedPropertyId) params.set('propertyId', selectedPropertyId)
   314→      const res = await fetch(`/api/tenants?${params.toString()}`)
   315→      if (!res.ok) throw new Error('Failed to fetch tenants')
   316→      const data = await res.json()
   317→      const tenantList = Array.isArray(data) ? data : []
   318→      setTenants(tenantList)
   319→      // Initialize attendance rows
   320→      setAttendanceRows(
   321→        tenantList.map((t: Tenant) => ({
   322→          tenantId: t.id,
   323→          name: t.user?.name || t.name,
   324→          present: true,
   325→          guestCount: 0,
   326→          notes: '',
   327→        }))
   328→      )
   329→    } catch (error) {
   330→      console.error('Error fetching tenants:', error)
   331→    }
   332→  }, [selectedPropertyId])
   333→
   334→  const fetchInventoryItems = useCallback(async () => {
   335→    try {
   336→      const params = new URLSearchParams()
   337→      if (selectedPropertyId) params.set('propertyId', selectedPropertyId)
   338→      const res = await fetch(`/api/inventory?${params.toString()}`)
   339→      if (!res.ok) throw new Error('Failed to fetch inventory')
   340→      const data = await res.json()
   341→      setInventoryItems(Array.isArray(data.items) ? data.items : [])
   342→    } catch (error) {
   343→      console.error('Error fetching inventory:', error)
   344→    }
   345→  }, [selectedPropertyId])
   346→
   347→  // ─── Effects ───────────────────────────────────────────────
   348→  useEffect(() => {
   349→    fetchMessData(activeTab)
   350→  }, [activeTab, attendanceDate, consumptionDays, wasteDays, fetchMessData])
   351→
   352→  useEffect(() => {
   353→    fetchTenants()
   354→  }, [selectedPropertyId, fetchTenants])
   355→
   356→  useEffect(() => {
   357→    fetchInventoryItems()
   358→  }, [selectedPropertyId, fetchInventoryItems])
   359→
   360→  // Load existing attendance records into the rows when data changes
   361→  useEffect(() => {
   362→    if (messData.attendance && tenants.length > 0) {
   363→      const mealData = messData.attendance[selectedMeal]
   364→      if (mealData?.records?.length) {
   365→        setAttendanceRows(prev =>
   366→          prev.map(row => {
   367→            const existing = mealData.records.find(
   368→              (r: AttendanceRecord) => r.tenantId === row.tenantId
   369→            )
   370→            if (existing) {
   371→              return {
   372→                ...row,
   373→                present: existing.present,
   374→                guestCount: existing.guestCount,
   375→                notes: existing.notes || '',
   376→              }
   377→            }
   378→            return row
   379→          })
   380→        )
   381→      }
   382→    }
   383→  }, [messData.attendance, selectedMeal, tenants.length])
   384→
   385→  // ─── Computed values ───────────────────────────────────────
   386→  const attendanceStats = useMemo(() => {
   387→    const mealData = messData.attendance?.[selectedMeal]
   388→    if (!mealData) {
   389→      return { totalPresent: 0, totalGuests: 0, attendanceRate: 0, totalTenants: tenants.length }
   390→    }
   391→    const totalPresent = mealData.present
   392→    const totalGuests = mealData.guests
   393→    const total = mealData.present + mealData.absent
   394→    const rate = total > 0 ? Math.round((totalPresent / total) * 100) : 0
   395→    return { totalPresent, totalGuests, attendanceRate: rate, totalTenants: total }
   396→  }, [messData.attendance, selectedMeal, tenants.length])
   397→
   398→  const filteredConsumption = useMemo(() => {
   399→    const logs = messData.consumption || []
   400→    if (!consumptionSearch) return logs
   401→    const q = consumptionSearch.toLowerCase()
   402→    return logs.filter(
   403→      (c: ConsumptionLog) =>
   404→        c.item?.name?.toLowerCase().includes(q) ||
   405→        c.mealType?.toLowerCase().includes(q)
   406→    )
   407→  }, [messData.consumption, consumptionSearch])
   408→
   409→  const consumptionTotals = useMemo(() => {
   410→    return filteredConsumption.reduce(
   411→      (acc: { issued: number; consumed: number; returned: number; wastage: number; cost: number }, c: ConsumptionLog) => ({
   412→        issued: acc.issued + c.issuedQty,
   413→        consumed: acc.consumed + c.consumedQty,
   414→        returned: acc.returned + c.returnedQty,
   415→        wastage: acc.wastage + c.wastageQty,
   416→        cost: acc.cost + c.totalCost,
   417→      }),
   418→      { issued: 0, consumed: 0, returned: 0, wastage: 0, cost: 0 }
   419→    )
   420→  }, [filteredConsumption])
   421→
   422→  const consumptionChartData = useMemo(() => {
   423→    if (!messData.consumption?.length) return []
   424→    const grouped: Record<string, { consumed: number; wastage: number; label: string }> = {}
   425→    for (const log of messData.consumption) {
   426→      try {
   427→        const dayKey = format(new Date(log.date), 'dd MMM')
   428→        if (!grouped[dayKey]) grouped[dayKey] = { consumed: 0, wastage: 0, label: dayKey }
   429→        grouped[dayKey].consumed += log.consumedQty
   430→        grouped[dayKey].wastage += log.wastageQty
   431→      } catch {
   432→        // skip invalid dates
   433→      }
   434→    }
   435→    return Object.values(grouped).sort((a, b) => a.label.localeCompare(b.label))
   436→  }, [messData.consumption])
   437→
   438→  const wasteStats = useMemo(() => {
   439→    return messData.wasteStats || { totalWaste: 0, foodWaste: 0, expired: 0, damaged: 0 }
   440→  }, [messData.wasteStats])
   441→
   442→  // ─── Handlers ──────────────────────────────────────────────
   443→  const handleMarkAttendance = async () => {
   444→    if (!attendanceRows.length) return
   445→    setSavingAttendance(true)
   446→    try {
   447→      const res = await fetch('/api/mess', {
   448→        method: 'POST',
   449→        headers: { 'Content-Type': 'application/json' },
   450→        body: JSON.stringify({
   451→          type: 'attendance',
   452→          propertyId: selectedPropertyId,
   453→          date: format(attendanceDate, 'yyyy-MM-dd'),
   454→          mealType: selectedMeal,
   455→          userId: currentUser?.id,
   456→          records: attendanceRows.map(r => ({
   457→            tenantId: r.tenantId,
   458→            present: r.present,
   459→            guestCount: r.guestCount,
   460→            notes: r.notes,
   461→          })),
   462→        }),
   463→      })
   464→      if (!res.ok) throw new Error('Failed to mark attendance')
   465→      const data = await res.json()
   466→      toast({
   467→        title: 'Attendance Marked',
   468→        description: `Successfully marked attendance for ${data.count} tenant(s)`,
   469→      })
   470→      fetchMessData('attendance')
   471→    } catch (error) {
   472→      console.error('Error marking attendance:', error)
   473→      toast({ title: 'Error', description: 'Failed to mark attendance', variant: 'destructive' })
   474→    } finally {
   475→      setSavingAttendance(false)
   476→    }
   477→  }
   478→
   479→  const handleAddConsumption = async () => {
   480→    if (!consumptionForm.itemId) {
   481→      toast({ title: 'Validation Error', description: 'Please select an inventory item', variant: 'destructive' })
   482→      return
   483→    }
   484→    setSavingConsumption(true)
   485→    try {
   486→      const selectedItem = inventoryItems.find(i => i.id === consumptionForm.itemId)
   487→      const totalCost = (consumptionForm.consumedQty + consumptionForm.wastageQty) * (selectedItem?.unitPrice || 0)
   488→
   489→      const res = await fetch('/api/mess', {
   490→        method: 'POST',
   491→        headers: { 'Content-Type': 'application/json' },
   492→        body: JSON.stringify({
   493→          type: 'consumption',
   494→          propertyId: selectedPropertyId,
   495→          date: consumptionForm.date,
   496→          itemId: consumptionForm.itemId,
   497→          mealType: consumptionForm.mealType,
   498→          issuedQty: consumptionForm.issuedQty,
   499→          consumedQty: consumptionForm.consumedQty,
   500→          returnedQty: consumptionForm.returnedQty,
   501→          wastageQty: consumptionForm.wastageQty,
   502→          unit: selectedItem?.unit || 'kg',
   503→          costPerUnit: selectedItem?.unitPrice || 0,
   504→          totalCost,
   505→          notes: consumptionForm.notes,
   506→          userId: currentUser?.id,
   507→        }),
   508→      })
   509→      if (!res.ok) throw new Error('Failed to add consumption log')
   510→      toast({ title: 'Success', description: 'Consumption log added successfully' })
   511→      setShowConsumptionDialog(false)
   512→      setConsumptionForm({
   513→        itemId: '',
   514→        mealType: 'breakfast',
   515→        issuedQty: 0,
   516→        consumedQty: 0,
   517→        returnedQty: 0,
   518→        wastageQty: 0,
   519→        date: format(new Date(), 'yyyy-MM-dd'),
   520→        notes: '',
   521→      })
   522→      fetchMessData('consumption')
   523→    } catch (error) {
   524→      console.error('Error adding consumption:', error)
   525→      toast({ title: 'Error', description: 'Failed to add consumption log', variant: 'destructive' })
   526→    } finally {
   527→      setSavingConsumption(false)
   528→    }
   529→  }
   530→
   531→  const handleAddWaste = async () => {
   532→    if (!wasteForm.description) {
   533→      toast({ title: 'Validation Error', description: 'Please enter a description', variant: 'destructive' })
   534→      return
   535→    }
   536→    setSavingWaste(true)
   537→    try {
   538→      const res = await fetch('/api/mess', {
   539→        method: 'POST',
   540→        headers: { 'Content-Type': 'application/json' },
   541→        body: JSON.stringify({
   542→          type: 'waste',
   543→          propertyId: selectedPropertyId,
   544→          date: wasteForm.date,
   545→          category: wasteForm.category,
   546→          itemId: wasteForm.itemId || null,
   547→          description: wasteForm.description,
   548→          quantity: wasteForm.quantity,
   549→          unit: wasteForm.unit,
   550→          estimatedCost: wasteForm.estimatedCost,
   551→          disposalMethod: wasteForm.disposalMethod,
   552→          notes: wasteForm.notes,
   553→          userId: currentUser?.id,
   554→        }),
   555→      })
   556→      if (!res.ok) throw new Error('Failed to add waste record')
   557→      toast({ title: 'Success', description: 'Waste record added successfully' })
   558→      setShowWasteDialog(false)
   559→      setWasteForm({
   560→        category: 'food_waste',
   561→        description: '',
   562→        itemId: '',
   563→        quantity: 0,
   564→        unit: 'kg',
   565→        estimatedCost: 0,
   566→        disposalMethod: 'trash',
   567→        date: format(new Date(), 'yyyy-MM-dd'),
   568→        notes: '',
   569→      })
   570→      fetchMessData('waste')
   571→    } catch (error) {
   572→      console.error('Error adding waste:', error)
   573→      toast({ title: 'Error', description: 'Failed to add waste record', variant: 'destructive' })
   574→    } finally {
   575→      setSavingWaste(false)
   576→    }
